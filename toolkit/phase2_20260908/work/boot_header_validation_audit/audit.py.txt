def runtime(off):return 0x60080000+off-0xa34
def digest_image(b):
    ks,ka,rs,ra,ss,sa,tags,page=struct.unpack_from('<8I',b,8)
    ro=page+((ks+page-1)//page)*page
    so=ro+((rs+page-1)//page)*page
    h=hashlib.sha1()
    parts=[b[page:page+ks],b[8:12],b[ro:ro+rs],b[16:20],b[so:so+ss],b[24:28],b[32:36],b[36:40],b[40:48],b[48:64],b[64:576]]
    for part in parts:h.update(part)
    return h.digest()+bytes(12),dict(kernel_size=ks,kernel_addr=ka,ramdisk_size=rs,ramdisk_addr=ra,second_size=ss,second_addr=sa,tags_addr=tags,page_size=page,ramdisk_offset=ro)
