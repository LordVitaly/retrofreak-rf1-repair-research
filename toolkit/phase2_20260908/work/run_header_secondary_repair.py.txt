def read(path, lba, count):
    assert not path.exists()
    assert (lba,count) in ((0x7effff,1),(0x7efffe,1),(0x7f0000,209),(0x7f00d1,2)) or (0x800000 <= lba < 0x801000 and 0 < count <= 16 and lba+count <= 0x801000)

def read_window(d, sectors, prefix):
    assert 0 < sectors <= 4096
    data = bytearray()
    for offset in range(0, sectors, 16):
        part = read(d / f'{prefix}_{offset:04d}.raw', 0x800000+offset, min(16,sectors-offset))
        assert all(part[j+512:j+528] == bytes(16) for j in range(0,len(part),528))
        data.extend(b''.join(part[j:j+512] for j in range(0,len(part),528)))
    return bytes(data)
