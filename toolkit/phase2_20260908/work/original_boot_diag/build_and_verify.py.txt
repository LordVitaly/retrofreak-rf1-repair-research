def addr(off):
    return 0x60080000 + off - 0xa34

def branch_target(raw, off, pc, opcode):
    word = struct.unpack_from('<I', raw, off)[0]
    assert word >> 24 == opcode
    disp = word & 0xffffff
    if disp & 0x800000:
        disp -= 0x1000000
    return pc + 8 + disp * 4
