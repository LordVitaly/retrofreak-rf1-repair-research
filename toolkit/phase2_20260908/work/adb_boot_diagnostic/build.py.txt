def parse_cpio(data):
    rows=[];pos=0
    while True:
        assert data[pos:pos+6]==b'070701'
        head=data[pos:pos+110]
        fields=[int(head[6+8*i:14+8*i],16) for i in range(13)]
        nsz=fields[11];fsz=fields[6]
        name_raw=data[pos+110:pos+110+nsz]
        assert name_raw[-1:]==b'\0'
        name=name_raw[:-1].decode('ascii')
        body_off=align(pos+110+nsz,4);end=align(body_off+fsz,4)
        rows.append(dict(name=name,head=head,fields=fields,name_raw=name_raw,
                         name_pad=data[pos+110+nsz:body_off],
                         body=data[body_off:body_off+fsz],body_pad=data[body_off+fsz:end]))
        pos=end
        if name=='TRAILER!!!':break
    assert not any(data[pos:])
    return rows,data[pos:]

def pack_cpio(replacements):
    pieces=[]
    for row in rows:
        body=replacements.get(row['name'],row['body'])
        head=bytearray(row['head'])
        if body!=row['body']:
            head[54:62]=f'{len(body):08x}'.encode()
            body_pad=bytes((-len(body))%4)
        else:body_pad=row['body_pad']
        pieces.extend((head,row['name_raw'],row['name_pad'],body,body_pad))
    result=b''.join(pieces)
    return result+bytes((-len(result))%512)
