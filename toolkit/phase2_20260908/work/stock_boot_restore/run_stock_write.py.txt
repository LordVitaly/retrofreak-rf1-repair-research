def sha(data):
    return hashlib.sha256(data).hexdigest()


def getjson(path):
    return json.loads(Path(path).read_text())


def pin(path, digest):
    data = Path(path).read_bytes()
    assert sha(data) == digest, ('SHA mismatch', str(path))
    return data


def validate_stage(raw, kind, mode=None):
    assert len(raw) == 110352 and raw[110080:] == bytes(272)
    assert zlib.crc32(raw[:110080]) == 0xad0dadee
    assert raw[0xb408:0xb40c] == bytes.fromhex('0101a0e3')
    assert raw[0xaf60:0xaf64] == bytes.fromhex('0003a0e1')
    if mode is not None:
        assert raw[0x76f8:0x76fc] == struct.pack('<I', 0xe3a00000 | mode)
    denials = (0x1934, 0x3b0c, 0x5b60, 0x7038, 0x7dcc, 0xbd5c)
    if kind not in ('dry', 'write'):
        denials += (0xba10, 0xbc1c)
    for offset in denials:
        assert raw[offset:offset+8] == bytes.fromhex('0000e0e31eff2fe1')
    if kind != 'write':
        for offset in (0xbbc8, 0xbc90):
            assert raw[offset:offset+4] == bytes.fromhex('feffffea')

    current = pin(plan['current_boot']['path'], plan['current_boot']['sha256'])

    assert sha(old_main) == block['old_main_sha256'] and zlib.crc32(old_main) == int(block['old_main_crc32'], 16)
    backup_result = json.loads(pin(plan['native_capture_result_path'], plan['native_capture_result_sha256']))
    assert backup_result['all_pass']
    summary = json.loads(pin(target['backup_summary_path'], target['backup_summary_sha256']))

    assert sha(aux) == summary['aux16_sha256'] == block['native_aux_u16_sha256'] == target['backup_aux16_sha256']

        sector = physical * 4096

    written = set()

    assert struct.unpack_from('<2I',previous,110080) == (0x44475453,0x600982e4)
    assert previous[110348:] == bytes(4)
    expected_reads = {}
    for mode in (0, 1):
        owner = maps[mode][logical]
        assert owner > 0
        expected_reads[mode] = source if owner in written | {physical} else old_main
    out = BASE / 'writes' / f'block{physical}'

    out.mkdir(exist_ok=False)

    marker = b'RF1CKP02' + struct.pack('<IIII',110080,0xad0dadee,1,0x60000004) + bytes(504)

    def save(path, value):
        with path.open('x') as handle:
            json.dump(value, handle, indent=2)
            handle.flush()
            os.fsync(handle.fileno())

    def update(value):
        temp = statefile.with_suffix('.tmp')
        with temp.open('w') as handle:
            json.dump(value, handle, indent=2)
            handle.flush()
            os.fsync(handle.fileno())
        os.replace(temp, statefile)

    def read(path, lba, count):
        assert not path.exists()
        assert (lba,count) in ((0x7effff,1),(0x7efffe,1),(0x7f0000,209),(0x7f00d1,2)) or (0x800000 <= lba < 0x801000 and 0 < count <= 16 and lba+count <= 0x801000)
        console.command(f'RS {hex(lba)} {hex(count)} {path}')
        data = path.read_bytes()
        assert len(data) == count * 528
        return data

    def read_window(directory, sectors, prefix):
        data = bytearray()
        assert 0 < sectors <= 4096
        for offset in range(0, sectors, 16):
            part = read(directory / f'{prefix}_{offset:04d}.raw', 0x800000+offset, min(16,sectors-offset))
            assert all(part[j+512:j+528] == bytes(16) for j in range(0,len(part),528))
            data.extend(b''.join(part[j:j+512] for j in range(0,len(part),528)))
        return bytes(data)

    def load_window(directory, data, prefix):
        assert len(data) == BLOCK_BYTES
        for offset in range(0, len(data), 8192):
            path = directory / f'{prefix}_{offset//512:04d}.bin'
            path.write_bytes(data[offset:offset+8192])
            console.command(f'WS {hex(0x800000+offset//512)} {path}')

    def execute(kind):
        nonlocal previous
        directory = out / kind
        directory.mkdir()
        raw = payloads[kind]
        assert read(directory/'previous.raw',0x7f0000,209) == previous
        if kind in ('dry','write'):
            load_window(directory, source, 'source')
            assert read_window(directory,4096,'source_check') == source
        elif kind.startswith('reopen_mode'):
            mode = int(kind[-1])
            load_window(directory,bytes(value ^ 255 for value in expected_reads[mode]),'prefill')
        for number,offset in enumerate(range(0,110080,8192)):
            path = directory/f'part{number:02d}.bin'
            path.write_bytes(raw[offset:min(offset+8192,110080)])
            console.command(f'WS {hex(0x7f0000+offset//512)} {path}')
        assert read(directory/'ram_before.raw',0x7f0000,209) == raw
        if kind == 'write':
            assert result.get('dry_pass') and result['write_launches'] == 0
            save(out/'WRITE_ATTEMPTED_DO_NOT_REPEAT.json',dict(physical_block=physical,payload_sha256=sha(raw),source_sha256=sha(source),manifest_sha256=manifest_pin,plan_sha256=sha(plan_bytes)))
            result['write_launches'] = 1
        save(directory/'DISPATCHED_NO_RETRY.json',dict(kind=kind,payload_sha256=sha(raw)))
        update(dict(valid=False,phase=f'stock{physical}_{kind}_dispatched',output=str(out)))
        console.command('RD')
        assert read(directory/'marker.raw',0x7effff,1) == marker
        after = read(directory/'ram_after.raw',0x7f0000,209)
        status = read(directory/'status.raw',0x7f00d1,2)
        result['executions'].append(dict(kind=kind,ram_sha256=sha(after),status_hex=status[:64].hex()))
        assert after[:110080] == raw[:110080] and after[110348:] == bytes(4)
        assert struct.unpack_from('<2I',after,110080) == (0x44475453,0x600982e4)
        previous = after
        return directory,status

    def check_writer(directory,status,kind):
        meta = manifest['variants'][f'{physical}_{kind}']
        actual = list(struct.unpack_from('<16I',status,0))
        guard = list(struct.unpack_from('<8I',read_window(directory,1,'guard'),0))
        save(directory/'gate_result.json',dict(actual_status=actual,expected_status=meta['expected_status_u32'],actual_guard=guard,expected_guard=meta['expected_guard_u32']))
        assert actual == meta['expected_status_u32']
        assert guard == meta['expected_guard_u32']

    try:
        console = RamConsole(out)
        console.select()
        assert read(out/'marker.raw',0x7effff,1) == marker
        assert read(out/'window.raw',0x7efffe,1) == window
        directory,status = execute('dry')
        check_writer(directory,status,'dry')
        result['dry_pass'] = True
        print(f'physical{physical}: dry native256-page preflight PASS; zero native writes',flush=True)
        directory,status = execute('write')
        check_writer(directory,status,'write')
        result['immediate_main_and_native_aux_exact'] = True
        print(f'physical{physical}: one erase,256 programs,256 exact native main+aux readbacks',flush=True)
        for mode in (0,1):
            directory,status = execute(f'capture_mode{mode}')
            assert status[:8] == b'RF1READ1' and struct.unpack_from('<4I',status,8) == (0x6020,1,0,0xa8000)
            snapshot = read_window(directory,1344,'ftl')
            (directory/'data.bin').write_bytes(snapshot)
            mapping = list(struct.unpack_from('<1896H',snapshot,0xf00))
            changes = [dict(logical=n,before=a,after=b) for n,(a,b) in enumerate(zip(maps[mode],mapping)) if a != b]
            result['map_checks'][str(mode)] = dict(snapshot_sha256=sha(snapshot),map_changes=changes,target_physical=mapping[logical])
            assert not changes,('Mapping changed',mode,changes)
        result['maps_unchanged'] = True
        for mode in (0,1):
            directory,status = execute(f'reopen_mode{mode}')
            assert status[:8] == b'RF1READ2' and struct.unpack_from('<4I',status,8) == (logical*4096,4096,0,4096)
            data = read_window(directory,4096,'readback')
            (directory/'data.bin').write_bytes(data)
            expected = expected_reads[mode]
            result['readbacks'][str(mode)] = dict(physical_owner=maps[mode][logical],sha256=sha(data),expected_sha256=sha(expected),exact=data==expected,expected_kind='stock' if expected==source else 'current_diagnostic')
            assert data == expected,('Full logical read mismatch',mode,logical)
            print(f'physical{physical}: mode{mode} logical{logical} full2MiB exact',flush=True)
        result['both_modes_readback_exact'] = True
        result['all_pass'] = True
        update(dict(valid=True,ram_path=str(directory/'ram_after.raw'),ram_sha256=sha(previous),last_case=f'stock_physical{physical}_verified_mode1',batch=f'stock_write_block{physical}'))
    except BaseException as exc:
        result['error'] = repr(exc)
        update(dict(valid=False,phase=f'stock_physical{physical}_failed',error=repr(exc),output=str(out)))
        raise
    finally:

            console.close()
            result['commands'] = console.commands
        save(out/'result.json',result)
        print(json.dumps({key:value for key,value in result.items() if key not in ('commands','executions')},indent=2),flush=True)


if __name__ == '__main__':
    main()
