"""PC-only ARM replay. All NAND write calls terminate before execution."""
from pathlib import Path

from emulate_flashboot_bbt import DumpReplay
from emulate_flashboot_offline import FLASH,FTL,REGS,address,UC_ARM_REG_PC,UC_ARM_REG_LR,UC_ARM_REG_R0

class Replay(DumpReplay):
    def __init__(self,record):
        super().__init__(record)
        self.write_entries.remove(address(0x7038)) # only allow PC-side formatter
        self.attempted_writes=[]
    def read_raw(self,sector,count):
        block,local=divmod(sector,4096)
        if block in (12,13):
            p=OLD/'work/bbt_copy12_restore_attempt01'/f'final_block{block}.raw'
            return p.read_bytes()[local*528:(local+count)*528]
        return super().read_raw(sector,count)
    def code_hook(self,uc,pc,size,unused):
        if self.dump_mode and pc==address(0x9ef4):
            sector,main,aux,count=[uc.reg_read(r) for r in REGS]
            if not (0<count<=16 and 0<=sector and sector+count<=8388608):
                self.reads.append(dict(sector=sector,count=count,assumed_error='outside physical NAND'))
                uc.reg_write(UC_ARM_REG_R0,0xffffffff)
                uc.reg_write(UC_ARM_REG_PC,uc.reg_read(UC_ARM_REG_LR))
                return
        if pc==address(0x3b0c):
            sector,main,aux,mode=[uc.reg_read(r) for r in REGS]
            self.attempted_writes.append(dict(sector=sector,mode=mode,
                main_first64=bytes(uc.mem_read(main,64)).hex(),
                main4096_matches_current=bytes(uc.mem_read(main,4096))==b''.join((HERE/'bbt_cold_attempt01/block12_full.raw').read_bytes()[i:i+512] for i in range(0,8*528,528)),
                auxiliary_halfwords=[struct.unpack('<H',uc.mem_read(aux+i*16,2))[0] for i in range(8)]))
        super().code_hook(uc,pc,size,unused)
