"""Local NAND header geometry audit; conditional static evaluation, no hardware."""
import hashlib
import json
import struct
from pathlib import Path
root = Path(__file__).resolve().parents[1]
d = (root / 'work/FlashBoot.packet512.STATIC_ONLY.bin').read_bytes()
assert hashlib.sha256(d).hexdigest() == '<REDACTED:HASH_OR_BYTES>'
table = []
for offset in range(0x1a8ac, 0x1a8ac + 34*8, 8):
    code, flags, capacity = d[offset], d[offset+1], struct.unpack_from('<I', d, offset+4)[0]
    if 0x44 in (code, code+0x10):
        table.append(dict(offset=hex(offset), code=code, flags=flags, sectors=capacity))
copies = []
with (root / 'work/nand_native_ecc40/NAND_00.RAW').open('rb') as f:
    for block in (1,3,5,6):
        f.seek(block*256*8448)
        raw = f.read(8448)
        header = raw[528:1040]
        assert raw[:4] == bytes.fromhex('3b8cdcfc')
        assert header[10:12] == b'RK'
        sectors_per_block = int.from_bytes(header[0x1ea:0x1ec], 'little')
        sectors_per_page = header[0x1ec]
        copies.append(dict(physical_block=block, header_sha256=hashlib.sha256(header).hexdigest(),
                           idb_reserved_count=int.from_bytes(header[:2],'little'),
                           timing_byte=header[0x1e9], sectors_per_block=sectors_per_block,
                           sectors_per_page=sectors_per_page, ecc_byte=header[0x1ed]))
assert all((x['sectors_per_block'],x['sectors_per_page'],x['ecc_byte']) == (4096,16,40) for x in copies)
assert len(table) == 0, 'Revisit missing-ID finding if the table changes'
capacity = 8388608  # Independently measured physical capacity, NOT this loader's result.
blocks = capacity // copies[0]['sectors_per_block']
report = dict(scope='Static IDB override and capacity arithmetic, not successful hardware initialization',
              flashboot_capacity_table_match=table, idb_copies=copies,
              independently_measured_capacity_sectors=capacity, expected_physical_blocks=blocks,
              flashboot_id_0x44_capacity_table_match_found=False,
              fallback_at_0xB404='writes 0x40000000 to first chip capacity; requires further full-path audit',
              calculation='gFlash+0x28 / ((gFlash+0x18) * gFlash+0x10), FtlVarInit 0x7508..0x7530',
              bbt_predicates_with_expected_not_proven_runtime_capacity=(1896 < blocks and 1896 >= blocks//2 and 4 <= blocks//4),
              limitations=['Assumes header read succeeds on hardware and takes observed IDB override',
                           'Assumes only the single NAND chip previously identified',
                           'No proof of absence of FTL startup writes; no loader executed'])
with (root/'outputs/flashboot_idb_geometry_audit.json').open('x',encoding='utf-8') as f:
    json.dump(report,f,indent=2)
print(json.dumps(report,indent=2))
