"""Read saved FlashInfo fields from acquired IDB samples, not the console."""
import hashlib,json,struct
from pathlib import Path
root=Path(__file__).resolve().parents[1]
rows=[]
for b in (1,3,5,6):
    raw=(root/f'work/idb_ecc24_targeted_attempt1/block{b:02d}_page15.raw').read_bytes()
    # IDB header points at sector 0xF4: physical page15, lane4.
    p=raw[4*528:4*528+512]
    u16=lambda o:struct.unpack_from('<H',p,o)[0]
    u32=lambda o:struct.unpack_from('<I',p,o)[0]
    gates=dict(magic=u32(0x10)==0x4e414e44,length=u16(0x14)==0x120,
               manufacturer=p[0x24]==4,interface_field=p[0x1c]==1)
    sectors=u32(0x38)
    perblock=u32(0x28)*p[0x20]
    rows.append(dict(block=b,parameter_record_sha256=hashlib.sha256(p).hexdigest(),
                     gates=gates,all_tested_gates_pass=all(gates.values()),
                     sectors=sectors,sectors_per_ftl_block=perblock,
                     ftl_block_count=(sectors//perblock)&0xffff if perblock else None,
                     page_sectors=p[0x25],ecc=p[0x22]))
report=dict(scope='Saved parameter path supported by local fresh samples, not executed FTL init',rows=rows,
            path='FlashInit B5E0/B608 -> A158; A204..A210 copies record+0x18 length0x78 to gFlash+8. gFlash+0x28 thus comes from record+0x38.',
            caveats=['No claim of full parameter integrity or safe FTL startup','Hardware ECC status unavailable'])
with (root/'outputs/live_idb_flash_parameter_audit.json').open('x') as f:json.dump(report,f,indent=2)
print(json.dumps(report,indent=2))
