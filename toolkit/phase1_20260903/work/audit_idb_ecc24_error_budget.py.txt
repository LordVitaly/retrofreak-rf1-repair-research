"""Count observed differences; reference is a hypothesis, not data to flash."""
import json
from pathlib import Path
root=Path(__file__).resolve().parents[1]
ref=(root/'work/FlashBoot.packet512.STATIC_ONLY.bin').read_bytes()
rows=[]
for block in (1,3,5,6):
    d=(root/f'work/IDB_old_block{block:02d}_FlashBoot.CANDIDATE.bin').read_bytes()
    a,b=d[0x1a800:0x1ac00],ref[0x1a800:0x1ac00]
    count=sum((x^y).bit_count() for x,y in zip(a,b))
    rows.append(dict(block=block,physical_page=15,lanes=[0,1],bytes_compared=len(a),
                     mismatching_bits=count,exceeds_24=count>24,
                     bits_0_to_1_vs_reference=sum((x & (~y & 255)).bit_count() for x,y in zip(a,b)),
                     bits_1_to_0_vs_reference=sum((y & (~x & 255)).bit_count() for x,y in zip(a,b))))
report=dict(scope='Offline difference counting, not measured hardware BCH error counts',rows=rows,
            hypothesis='All four 1024-byte tail units exceed 24 differing bits relative to official. Consistent with an uncorrectable ECC24 unit if official is original and stored parity/layout use ECC24.',
            limitations=['Stored parity is not validated by this calculation',
                         'RS completion does not expose per-unit hardware ECC status',
                         'Changing decoder to ECC40 cannot correct parity originally encoded for ECC24',
                         'Reference changes versus original cannot be excluded by difference count alone'],
            device_actions='none',image_written=False)
with (root/'outputs/idb_ecc24_error_budget.json').open('x') as f:json.dump(report,f,indent=2)
print(json.dumps(report,indent=2))
