for rom,fill in itertools.product((0,0x56313030),(0,0xa5)):
    u=Uc(UC_ARCH_ARM,UC_MODE_ARM,cpu=UC_CPU_ARM_CORTEX_A9)
    for base,size in [(0x60000000,0x20000),(0x60080000,0x20000),(0x60200000,0x600000),(0x61000000,0x20000),(0x61200000,0x10000),(0,0x1000),(0x10080000,0x10000)]:u.mem_map(base,size)
    u.mem_write(0x60000000,bytes([fill])*0x20000)
    u.mem_write(0x60000000,blob)
    u.mem_map(0x10102000,0x1000,UC_PROT_READ)
    u.mem_write(0x101027fc,struct.pack('<I',rom))
    u.mem_map(0x20004000,0x1000,UC_PROT_WRITE)
    u.mem_map(0x20008000,0x1000,UC_PROT_WRITE)
    if args.timer_write_sink:u.mem_map(0x20038000,0x1000,UC_PROT_WRITE)
    if args.serial_write_sink:u.mem_map(0x20064000,0x1000,UC_PROT_WRITE)
    serial_sequence=[(0x20008074,0xf0005)] if rom==0x56313030 else [(0x200080bc,0x50005)]
    serial_sequence += [(0x20064000+offset,value) for offset,value in [(0x88,7),(4,0),(0x10,0),(0xc,0x83),(0,0xd),(4,0),(0xc,3),(0x98,1),(0x9c,3),(0xa0,1)]]
    serial_writes=[]
    peripheral=dict(timer_reads=0,uart_status_reads=0,uart_bytes=[])
    if args.synthetic_clock_uart:
        u.mem_protect(0x20038000,0x1000,UC_PROT_READ|UC_PROT_WRITE)
        u.mem_protect(0x20064000,0x1000,UC_PROT_READ|UC_PROT_WRITE)
        def peripheral_read(uc,access,addr,size,value,ctx):
            assert size==4 and addr in (0x20038004,0x2006407c),('unexpected peripheral read',hex(addr),size)
            if addr==0x20038004:
                # Deliberately synthetic; not calibrated wall time or RF-1 data.
                peripheral['timer_reads']+=1
                supplied=(0xffffffff-24000*peripheral['timer_reads'])&0xffffffff
            else:
                peripheral['uart_status_reads']+=1
                supplied=2
            uc.mem_write(addr,struct.pack('<I',supplied))
        u.hook_add(UC_HOOK_MEM_READ,peripheral_read,begin=0x20038000,end=0x20038fff)
        u.hook_add(UC_HOOK_MEM_READ,peripheral_read,begin=0x20064000,end=0x20064fff)
    state=dict(entries=0,phase='bootstrap',rom_reads=0,mmio_writes=[])
    def entry(uc,pc,size,ctx):
        state['entries']+=1
        if state['entries']==2:
            assert bytes(uc.mem_read(0x60000000,len(d)))==d
            state['phase']='flashboot'
            uc.mem_protect(0x60000000,0x20000,UC_PROT_READ|UC_PROT_EXEC)
    def read(uc,access,addr,size,value,ctx):
        assert addr==0x101027fc and size==4,('unapproved ROM read',hex(addr))
        state['rom_reads']+=1
    def write(uc,access,addr,size,value,ctx):
        if state['phase']=='bootstrap':
            ranges=[(0x60000000,0x60000000+len(d)),(0x61000000,0x61000000+len(blob)),(0x61010000,0x61018008),(0x61200000,0x61210000)]
        elif state['phase']=='main':
            # Broad data/stack ranges for boundary discovery, not a RAM safety proof.
            ranges=[(0x60099000,0x600a0000),(0x60200000,0x60800000)]
            if args.timer_write_sink and addr in (0x20038000,0x20038008):
                assert size==4 and value=={0x20038000:0,0x20038008:1}[addr]
                state['mmio_writes'].append(dict(address=hex(addr),value=hex(value)))
                return
            if args.serial_write_sink and len(serial_writes)<len(serial_sequence) and (addr,value)==serial_sequence[len(serial_writes)]:
                assert size==4
                serial_writes.append((addr,value))
                state['mmio_writes'].append(dict(address=hex(addr),value=hex(value)))
                return
            if args.synthetic_clock_uart and len(serial_writes)==len(serial_sequence) and addr==0x20064000:
                assert size==4 and value<=255 and uc.reg_read(UC_ARM_REG_PC)==0x60080000+0xad4-0xa34
                peripheral['uart_bytes'].append(value)
                return
        else:
            ranges=[(0,0x724),(0x60080000,0x6009a488)]

                assert size==4 and value==expected
                state['mmio_writes'].append(dict(address=hex(addr),value=hex(value)))
                return
