report=[]
for logical in (187,259):

    assert hashlib.sha256(raw).hexdigest()==build['sha256']
    for fail in (None,0,15):

        u.mem_write(0x62fff000,b'\x69'*0x202000);u.reg_write(UC_ARM_REG_R4,0x6009a1f8);calls=[]
        pattern=hashlib.shake_256(b'portion read data').digest(build['sectors']*512)
        def read(uc,pc,size,ctx):
            n=len(calls);args=[uc.reg_read(r) for r in (h.UC_ARM_REG_R0,h.UC_ARM_REG_R1,h.UC_ARM_REG_R2,h.UC_ARM_REG_R3)]
            assert args==[0,build['lba']+n*128,min(128,build['sectors']-n*128),0x63000000+n*65536]
