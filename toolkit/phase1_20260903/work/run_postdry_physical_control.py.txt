class PhysicalConsole(RamConsole):

    def command(self,value):
        parts=value.split();op=parts[0]
        assert op in ('DB','RFI','RS')
        if op=='RS':
            assert int(parts[1],16) in sectors and parts[2]=='0x10'

        if op=='DB':assert value==f'DB {loader}' and not any(x.startswith('DB ') for x in self.commands)

        self.commands.append(value);start=len(''.join(self.chunks));self.proc.write(value+'\r\n')

        while time.monotonic()<deadline:
            response=completed_response(''.join(self.chunks),start,token)
            if response is not None:return response
            if self.done.is_set():break
            time.sleep(.05)
