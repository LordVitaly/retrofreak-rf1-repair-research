def compensate(data,at,target):
    data[at:at+4]=bytes(4)
    base=zlib.crc32(data);basis={}
    for bit in range(32):
        trial=bytearray(data);trial[at+bit//8]^=1<<(bit%8)
        v=zlib.crc32(trial)^base;mask=1<<bit
        while v:
            p=v.bit_length()-1
            if p not in basis:basis[p]=(v,mask);break
            v^=basis[p][0];mask^=basis[p][1]
    assert len(basis)==32
    v=target^base;solution=0
    while v:
        p=v.bit_length()-1;v^=basis[p][0];solution^=basis[p][1]
    data[at:at+4]=struct.pack('<I',solution)
    assert zlib.crc32(data)==target
