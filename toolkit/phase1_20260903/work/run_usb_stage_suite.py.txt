    def reader(self):
        try:
            while self.proc.isalive():
                s=self.proc.read(4096)
                if s:self.chunks.append(s);self.stream.write(s);self.stream.flush()
        except EOFError:pass
        finally:self.done.set()
    def wait(self,token,start=0):
        deadline=time.monotonic()+15
        while time.monotonic()<deadline:
            s=''.join(self.chunks)[start:]
            if token in s:return s
            if self.done.is_set():break
            time.sleep(.05)
        raise RuntimeError('Console unavailable; no retry')
    def command(self,cmd):
        op=cmd.split()[0];assert op in ('DB','RS','WS','RD')
        token={'DB':'Download boot ok','RS':'Read Sector to file (100%)','WS':'Write Sector from file (100%)','RD':'Reset Device OK'}[op]
        result['commands'].append(cmd);start=len(''.join(self.chunks));self.proc.write(cmd+'\r\n')
        deadline=time.monotonic()+(45 if op=='DB' else 6 if op=='RD' else 15)
        while time.monotonic()<deadline:
            response=completed_response(''.join(self.chunks),start,token)
            if response is not None:return response
            if self.done.is_set():break
            time.sleep(.05)
        raise RuntimeError('No command completion; no retry')
    def close(self):
        if self.proc.isalive():self.proc.terminate(force=True)
        self.done.wait(2)
