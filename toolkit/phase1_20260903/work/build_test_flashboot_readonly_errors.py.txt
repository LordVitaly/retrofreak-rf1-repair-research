source=(ROOT/'work/FlashBoot.packet512.STATIC_ONLY.bin').read_bytes()
assert hashlib.sha256(source).hexdigest()=='<REDACTED:HASH_OR_BYTES>'
entries={0x1934:'backend service-write dispatch',0x3b0c:'FTL physical write wrapper',0x5b60:'NAND service-write cache',0x7038:'format path',0x7dcc:'logical writer',0xba10:'physical program',0xbc1c:'physical erase',0xbd5c:'physical copy'}
stub=bytes.fromhex('0000e0e31eff2fe1') # MVN r0,#0; BX lr (ARM)
candidate=bytearray(source)
patches=[]
for off,label in entries.items():
    candidate[off:off+8]=stub
    patches.append(dict(offset=hex(off),purpose=label,before=source[off:off+8].hex(),after=stub.hex()))
assert source[0x18bf8:0x18bfc]==bytes.fromhex('080094e5')
candidate[0x18bf8:0x18bfc]=bytes.fromhex('410000ea') # existing USB-only gate
allowed={o+i for o in entries for i in range(8)}|set(range(0x18bf8,0x18bfc))
assert all(i in allowed for i,(a,b) in enumerate(zip(source,candidate)) if a!=b)
tests=[]
for off in entries:
    for value in (0,0xffffffff,0x12345678):
        u=Uc(UC_ARCH_ARM,UC_MODE_ARM)
        u.mem_map(0x60080000,0x20000,UC_PROT_READ|UC_PROT_EXEC)
        u.mem_write(0x60080000,bytes(candidate[0xa34:]))
        for reg in REGS:u.reg_write(reg,value)
        u.reg_write(UC_ARM_REG_SP,0xdead0000) # deliberately unmapped
        stop=0x6007f000
        u.reg_write(UC_ARM_REG_LR,stop)
        writes=[]
        u.hook_add(UC_HOOK_MEM_WRITE,lambda uc,a,p,s,v,c:writes.append((p,s,v)))
        u.emu_start(address(off),stop,count=3)
        assert u.reg_read(UC_ARM_REG_PC)==stop and u.reg_read(UC_ARM_REG_R0)==0xffffffff
        assert u.reg_read(UC_ARM_REG_SP)==0xdead0000 and not writes
        assert all(u.reg_read(reg)==value for reg in REGS[1:])
        tests.append(dict(entry=hex(off),argument=value,error_return='0xffffffff',guest_writes=0))

class ErrorReplay(RemapReplay):
    def __init__(self,record):
        super().__init__(record)
        self.write_entries.clear() # actual patched instructions, not host return stubs
        self.hits=[]
        self.guard_addresses={address(o) for o in entries}
        self.uc.mem_write(0x60080000,bytes(candidate[0xa34:]))
    def code_hook(self,uc,pc,size,unused):
        if pc in self.guard_addresses:self.hits.append(hex(pc-0x60080000+0xa34))
        super().code_hook(uc,pc,size,unused)
