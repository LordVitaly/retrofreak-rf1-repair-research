"""Register/reset-completion contract for common init; no card/NAND model."""
import struct
from unicorn import UC_HOOK_MEM_READ,UC_HOOK_MEM_WRITE
class MMCContract:
    def __init__(self,u):
        self.regs={};self.writes=[]
        for base in (0x10214000,0x10218000,0x1021c000):
            u.mem_map(base,0x1000)
            u.hook_add(UC_HOOK_MEM_READ,self.read,begin=base,end=base+0xfff)
            u.hook_add(UC_HOOK_MEM_WRITE,self.write,begin=base,end=base+0xfff)
    def read(self,u,access,a,size,value,ctx):
        off=a&0xfff;assert size==4 and off in (0,0x48,0x2c),(hex(a),size)
        val=self.regs.get(a,0)&~7 if off==0 else 0
        u.mem_write(a,struct.pack('<I',val))
    def write(self,u,access,a,size,val,ctx):
        off=a&0xfff
        assert size==4 and off in (0,4,0x78,0x4c,0x18,0xc,0x64,0x44,0x24),(hex(a),hex(val))
        # CMD(2c), data count, DMA descriptors and data/FIFO are forbidden.
        if off==0:assert val in (2,3,0,0x10),hex(val)
        self.regs[a]=val;self.writes.append((a,val))
