"""Fixed five-page physical read. Default is offline preparation only.

Execution requires fresh user-confirmed MaskROM; never runs FlashBoot or FTL.
"""
import argparse
import hashlib
import json
import os
import subprocess
import threading
import time
from pathlib import Path
from audit_ram_loader_container import BASE, crc32, rc4
from direct_winpty_console import DirectConsole
from read_device_identity import TOOL

ROOT=Path(__file__).resolve().parent
LOADER=BASE/'components/RetronLoader.bin'
SHA='<REDACTED:HASH_OR_BYTES>'
PAGES=((1,14),(1,15),(3,15),(5,15),(6,15))

def prepare():
    import struct
    d=LOADER.read_bytes()
    assert hashlib.sha256(d).hexdigest()==SHA
    assert crc32(d[:-4])==struct.unpack_from('<I',d,len(d)-4)[0]
    ref=(ROOT/'FlashBoot.packet512.STATIC_ONLY.bin').read_bytes()
    expected=ref[0x18800:0x1a800]
    assert len(expected)==8192
    for b in (1,3,5,6):
        assert (ROOT/f'IDB_old_block{b:02d}_FlashBoot.CANDIDATE.bin').read_bytes()[0x18800:0x1a800]==expected
    return expected

def decode_page(raw):
    if len(raw)!=8448: raise ValueError('Wrong page size')
    return b''.join(rc4(raw[i*528:i*528+512]) for i in range(16))

def execute(expected, parameters_only=False):
    # Only run after the operator confirms a new hardware MaskROM entry.
    check=subprocess.run(['powershell','-NoProfile','-Command',
        "@(Get-Process -Name Android_Console_Tool -ErrorAction SilentlyContinue).Count"],capture_output=True,text=True,check=True)
    if check.stdout.strip()!='0': raise RuntimeError('Another device tool is active')
    out=ROOT/('idb_ecc24_parameter_attempt2' if parameters_only else 'idb_ecc24_targeted_attempt1')
    out.mkdir(exist_ok=False)
    commands=['RID','RFI'] if parameters_only else [f'DB {LOADER}', 'RID','RFI']
    targets={}
    for block,page in PAGES:
        count=1 if parameters_only and page==15 else 16
        lane=4 if count==1 else 0
        path=out/f'block{block:02d}_page{page:02d}.raw'
        cmd=f'RS {hex(block*4096+page*16+lane)} {hex(count)} {path}'
        commands.append(cmd)
        targets[cmd]=(path,block,page,count)
    env=dict(os.environ)
    env['__COMPAT_LAYER']='RunAsInvoker'
    proc=DirectConsole.spawn([str(TOOL)],cwd=str(TOOL.parent),env=env,dimensions=(40,160),backend=1)
    chunks=[]
    done=threading.Event()
    log=(out/'transcript.txt').open('x',encoding='utf-8')
    def read():
        try:
            while proc.isalive():
                s=proc.read(4096)
                if s:
                    chunks.append(s);log.write(s);log.flush()
        except EOFError: pass
        finally: done.set()
    thread=threading.Thread(target=read,daemon=True)
    thread.start()
    def wait(fragment,start=0):
        deadline=time.monotonic()+45
        while time.monotonic()<deadline:
            text=''.join(chunks)
            if fragment in text[start:]:return text[start:]
            if done.is_set() or not proc.isalive():break
            time.sleep(.05)
        raise RuntimeError('Missing response; no next command')
    try:
        intro=wait('Select input')
        required=('Found 1 rockusb','DevNo=1') if parameters_only else ('Found 1 rockusb','DevNo=1','Maskrom')
        # USBplug may retain the Maskrom label after successful RAM DB.
        # No DB is sent in parameters-only mode: RID/RFI and exact physical
        # control data establish a working read path before parameter reads.
        if not all(x in intro for x in required):
            raise RuntimeError('Exactly one fresh Maskrom device required; no DB sent')
        start=len(''.join(chunks));proc.write('1\r\n');wait('Rockusb>',start)
        for cmd in commands:
            start=len(''.join(chunks));proc.write(cmd+'\r\n')
            response=wait('Rockusb>',start)
            if cmd.startswith('DB ') and 'Download boot ok' not in response:raise RuntimeError('DB failed')
            if cmd=='RID' and '2C 44 44 4B A9' not in response:raise RuntimeError('Unexpected ID')
            if cmd=='RFI' and not all(s in response for s in ('MICRON','4096MB','ECC Bits: 40','2048KB','8KB')):
                raise RuntimeError('Unexpected geometry')
            if cmd in targets:
                path,b,p,count=targets[cmd]
                if '(100%)' not in response or not path.exists() or path.stat().st_size!=count*528:
                    raise RuntimeError('Incomplete read')
                raw=path.read_bytes()
                record=dict(block=b,page=p,count=count,lane=4 if count==1 else 0,sha256=hashlib.sha256(raw).hexdigest(),hardware_ecc_status_available=False)
                if p==14:
                    record['control_exact']=decode_page(raw)==expected
                    if not record['control_exact']:raise RuntimeError('Control mismatch; tail reads cancelled')
                with (out/'manifest.jsonl').open('a') as f:f.write(json.dumps(record)+'\n')
                print(json.dumps(record),flush=True)
        proc.write('Q\r\n')
    finally:
        if proc.isalive():proc.terminate(force=True)
        done.wait(2)
        if done.is_set():log.close()

if __name__=='__main__':
    parser=argparse.ArgumentParser()
    parser.add_argument('--execute-after-user-confirmed-maskrom',action='store_true')
    parser.add_argument('--read-parameters-with-running-loader',action='store_true')
    args=parser.parse_args()
    expected=prepare()
    if args.execute_after_user_confirmed_maskrom and args.read_parameters_with_running_loader:parser.error('Choose one mode')
    if args.read_parameters_with_running_loader:execute(expected,parameters_only=True)
    elif args.execute_after_user_confirmed_maskrom:execute(expected)
    else:print(json.dumps(dict(mode='OFFLINE ONLY',loader_sha256=SHA,control_bytes=len(expected),pages=PAGES,device_commands_sent=0)))
