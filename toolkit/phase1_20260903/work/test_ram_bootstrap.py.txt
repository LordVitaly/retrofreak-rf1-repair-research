def run(name,payload,fill,expect_ok):
    u=Uc(UC_ARCH_ARM,UC_MODE_ARM,cpu=UC_CPU_ARM_CORTEX_A9)
    u.mem_map(0x60000000,0x20000)
    u.mem_write(0x60000000,bytes([fill])*0x20000)
    u.mem_write(0x60000000,payload)
    u.mem_map(0x61000000,0x20000)
    u.mem_write(0x61000000,bytes([fill])*0x20000)
    u.mem_map(0x61200000,0x10000,UC_PROT_READ|UC_PROT_WRITE)
    u.reg_write(UC_ARM_REG_SP,0xdead0000) # stub must establish its own stack
    state=dict(phase='copy',visits=0,failure_hits=0,violations=[],copied_exact=False)
    def entry(uc,address,size,ctx):
        state['visits']+=1
        if state['visits']==2: uc.emu_stop()
    def relocated(uc,address,size,ctx):
        state['copied_exact']=bytes(uc.mem_read(0x61000000,len(payload)))==payload
        state['phase']='decode'
        uc.mem_protect(0x61000000,0x10000,UC_PROT_READ|UC_PROT_EXEC)
    def failed(uc,address,size,ctx):
        state['failure_hits']+=1
        if state['failure_hits']==8: uc.emu_stop()
    def write(uc,access,address,size,value,ctx):
        ranges=([(0x61000000,0x61000000+len(payload))] if state['phase']=='copy' else
            [(0x60000000,0x60000000+len(source)),(symbols['__bss_start'],symbols['__bss_end']),(0x61200000,0x61210000)])
        if not any(lo<=address and address+size<=hi for lo,hi in ranges):
            state['violations'].append((hex(address),size)); uc.emu_stop()
    u.hook_add(UC_HOOK_CODE,entry,begin=0x60000000,end=0x60000000)
    u.hook_add(UC_HOOK_CODE,relocated,begin=symbols['relocated'],end=symbols['relocated'])
    u.hook_add(UC_HOOK_CODE,failed,begin=symbols['failure'],end=symbols['failure'])
    u.hook_add(UC_HOOK_MEM_WRITE,write)
    error=None
    try: u.emu_start(0x60000000,0,timeout=30000000,count=100000000)
    except UcError as e: error=str(e)
    pc=u.reg_read(UC_ARM_REG_PC)
    assert not error and not state['violations'] and state['copied_exact'],(name,error,state,hex(pc))
    output=bytes(u.mem_read(0x60000000,len(source)))
    if expect_ok: assert pc==0x60000000 and state['visits']==2 and output==source
    else: assert pc==symbols['failure'] and state['visits']==1 and state['failure_hits']==8
    assert bytes(u.mem_read(0x60000000+len(source),0x20000-len(source)))==bytes([fill])*(0x20000-len(source))
    assert bytes(u.mem_read(0x61000000,len(payload)))==payload
    result=dict(case=name,pc=hex(pc),fill=fill,returned_to_flashboot=expect_ok,decoder_result=None if expect_ok else u.reg_read(UC_ARM_REG_R0),output_sha256=hashlib.sha256(output).hexdigest(),state=state)
    print('PASS',name,flush=True)
    return result
