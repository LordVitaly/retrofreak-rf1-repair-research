"""One explicit RAM DB after user-confirmed fresh MaskROM. No NAND commands.

import argparse,hashlib,json,os,subprocess,threading,time
from pathlib import Path
from direct_winpty_console import DirectConsole
from read_device_identity import TOOL
ROOT=Path(__file__).resolve().parents[1]
p=argparse.ArgumentParser();p.add_argument('--execute-after-fresh-maskrom',action='store_true');p.add_argument('--usbplug-control',action='store_true');p.add_argument('--native-cpu-entry',action='store_true');p.add_argument('--plain-usb-control',action='store_true');args=p.parse_args()
assert not args.native_cpu_entry or args.usbplug_control
assert not args.plain_usb_control or not (args.native_cpu_entry or args.usbplug_control)
assert args.execute_after_fresh_maskrom,'Explicit hardware mode required'
loader=ROOT/'work/readonly_ram.BOOT.NOT_FOR_UPLOAD.bin'
expected_hash='<REDACTED:HASH_OR_BYTES>'
attempt='readonly_hardware_attempt01'
if args.usbplug_control:
    loader=ROOT/'work/usbplug_control_ram.BOOT.NOT_FOR_UPLOAD.bin'
    expected_hash='<REDACTED:HASH_OR_BYTES>'
    attempt='usbplug_control_hardware_attempt01'
    if args.native_cpu_entry:
        loader=ROOT/'work/usbplug_native_entry_ram.BOOT.NOT_FOR_UPLOAD.bin'
        expected_hash='<REDACTED:HASH_OR_BYTES>'
        attempt='usbplug_native_entry_hardware_attempt01'
if args.plain_usb_control:
    loader=ROOT/'work/plain_usb_61440_control.RAM_ONLY.bin'
    expected_hash='<REDACTED:HASH_OR_BYTES>'
    attempt='plain_usb_control_hardware_attempt01'
assert hashlib.sha256(loader.read_bytes()).hexdigest()==expected_hash
check=subprocess.run(['powershell','-NoProfile','-Command','@(Get-Process -Name Android_Console_Tool -ErrorAction SilentlyContinue).Count'],capture_output=True,text=True,check=True)
assert check.stdout.strip()=='0','Another console tool is active'
out=ROOT/'work'/attempt;out.mkdir(exist_ok=False)
env=dict(os.environ);env['__COMPAT_LAYER']='RunAsInvoker'
proc=DirectConsole.spawn([str(TOOL)],cwd=str(TOOL.parent),env=env,dimensions=(40,160),backend=1)
chunks=[];done=threading.Event();stream=(out/'transcript.txt').open('x',encoding='utf-8')
def reader():
    try:
        while proc.isalive():
            s=proc.read(4096)
            if s:chunks.append(s);stream.write(s);stream.flush()
    except EOFError:pass
    finally:done.set()
threading.Thread(target=reader,daemon=True).start()
def wait(fragment,start=0):
    deadline=time.monotonic()+45
    while time.monotonic()<deadline:
        text=''.join(chunks)
        if fragment in text[start:]:return text[start:]
        if done.is_set() or not proc.isalive():break
        time.sleep(.05)
    raise RuntimeError('Expected response absent; no next command sent')
result=dict(user_fresh_maskrom_confirmed=True,residual_risk_accepted=True,commands=[],nand_write_commands_sent=False)
try:
    intro=wait('Select input')
    assert all(s in intro for s in ('Found 1 rockusb','DevNo=1','Maskrom'))
    start=len(''.join(chunks));proc.write('1\r\n');wait('Rockusb>',start)
    command=f'DB {loader}'
    result['commands'].append(command)
    start=len(''.join(chunks));proc.write(command+'\r\n')
    response=wait('Rockusb>',start)
    result['download_reported_ok']='Download boot ok' in response
    result['response']=response
    print(json.dumps(result,indent=2),flush=True)
finally:
    if proc.isalive():proc.terminate(force=True)
    done.wait(2)
    if done.is_set():stream.close()
    (out/'result.json').write_text(json.dumps(result,indent=2)+'\n')
