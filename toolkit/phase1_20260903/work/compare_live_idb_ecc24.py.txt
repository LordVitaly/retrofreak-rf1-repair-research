"""Compare already acquired samples; no device actions."""
import json
from pathlib import Path
from idb_ecc24_targeted_read import decode_page
root=Path(__file__).resolve().parents[1]
ref=(root/'work/FlashBoot.packet512.STATIC_ONLY.bin').read_bytes()[0x1a800:]
rows=[]
for b in (1,3,5,6):
    raw=(root/f'work/idb_ecc24_targeted_attempt1/block{b:02d}_page15.raw').read_bytes()
    d=decode_page(raw)
    old=(root/f'work/IDB_old_block{b:02d}_FlashBoot.CANDIDATE.bin').read_bytes()[0x1a800:0x1ae00]
    rows.append(dict(block=b,first1024_bit_differences_vs_official=sum((a^c).bit_count() for a,c in zip(d[:1024],ref[:1024])),
                     first1536_byte_differences_vs_official=sum(a!=c for a,c in zip(d,ref)),
                     first1536_exact_old=d[:1536]==old,
                     first1536_bit_changes_since_old=sum((a^c).bit_count() for a,c in zip(d,old)),
                     device_code_entry23=hex(d[0x164])))
report=dict(scope='Fresh ECC24 physical RS after confirmed MaskROM and exact 8192-byte control; no NAND writes',
            rows=rows,hardware_ecc_status_available=False)
with (root/'outputs/live_idb_ecc24_comparison.json').open('x') as f:json.dump(report,f,indent=2)
print(json.dumps(report,indent=2))
