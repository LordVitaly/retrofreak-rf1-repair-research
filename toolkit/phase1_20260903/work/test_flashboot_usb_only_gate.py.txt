    u.mem_map(0,0x1000) # observe low-vector branch in the unpatched control
    u.mem_map(0x60080000,0x20000)
    u.mem_write(0x60080000,bytes(binary[0xa34:]))
    u.mem_map(0x60200000,0x600000)
    # Recreate the scalar state immediately before parameter parsing.

    u.mem_write(0x6009a200,struct.pack('<I',bootflag))
    u.mem_write(0x6009a208,struct.pack('<I',0))
