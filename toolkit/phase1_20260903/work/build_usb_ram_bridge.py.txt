def branch(at,target,cond=14,link=False):
    delta=target-(0x60000000+at+8)
    assert delta%4==0 and -(1<<25)<=delta<(1<<25)
    return struct.pack('<I',(cond<<28)|(0x0b000000 if link else 0x0a000000)|((delta//4)&0xffffff))
