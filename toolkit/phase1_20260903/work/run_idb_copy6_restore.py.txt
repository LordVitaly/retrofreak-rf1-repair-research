def save(name,obj):
    with (out/name).open('x',encoding='utf-8') as f:json.dump(obj,f,indent=2);f.flush();os.fsync(f.fileno())
def main(raw):
    assert len(raw)%528==0
    return b''.join(raw[i:i+512] for i in range(0,len(raw),528))
def cmd(value,token):
    assert value in ('RID','RFI') or value=='DB '+os.path.relpath(builddir/'writer.bin',out)
    start=len(''.join(c.chunks));c.commands.append(value);c.send_once(value)
    deadline=time.monotonic()+45
    while time.monotonic()<deadline:
        text=''.join(c.chunks);response=completed_response(text,start,token)
        if response is not None:return response
        plain=ANSI.sub('',text[start:])
        if 'Error:' in plain and 'Rockusb>' in plain:raise RuntimeError(plain)
        if c.done.is_set():break
        time.sleep(.005)
    raise RuntimeError('Missing command completion; no retry: '+value)
def rs(name,sector,count):
    assert (sector==0x7effff and count==1) or (0<=sector<0x12000 and 0<count<=4096 and sector+count<=0x12000)
    path=out/(name+'.raw');assert not path.exists()
    c.command(f'RS {hex(sector)} {hex(count)} {path}')
    raw=path.read_bytes();assert len(raw)==count*528
    return raw
def marker(name,phase,programs,next_sector,erases):
    raw=rs(name,0x7effff,1);w=list(struct.unpack('<18I',raw[:72]))
    save(name+'.json',dict(words=w))

    assert w[4:10]==[erases,0,programs,0,next_sector,0],w
    assert w[13:]==[1,build['build_tag'],1,0x5770,1],w
    return w
