        if bootsetup:previous_sha='<REDACTED:HASH_OR_BYTES>'
        if actualusb:previous_sha='<REDACTED:HASH_OR_BYTES>'
        assert hashlib.sha256(previous.read_bytes()).hexdigest()==previous_sha,'Previous exact hardware state lost: stop without upload'
        result['previous_hardware_state_exact']=True
    for (name,digest,cut),raw in zip(cases,payloads):
        case=dict(name=name,sha256=digest,cut_pc=hex(cut),returned=False)
        result['cases'].append(case)
        for i,offset in enumerate(range(0,110080,8192)):
            if entry_partial and name=='moved_helper_control' and i<2:continue
            if partial and name=='at_low_entry' and i<13:continue
            part=out/f'{name}_{i:02d}.bin'
            with part.open('xb') as f:f.write(raw[offset:min(offset+8192,110080)])
            command(f'WS {hex(0x7f0000+offset//512)} {part}')

        assert before.read_bytes()==raw,'RAM payload differs: no RD'

        if nandid:

            case['nand_controller_revision']=struct.unpack_from('<I',data,110268)[0]
        if geometry:case['nand_geometry_bytes']=data[110272:110344].hex()
        if firstread:case['first_read_return']=hex(struct.unpack_from('<I',data,110344)[0])
        if nandinit:case['nand_init_return']=case.pop('first_read_return')

            case['startup_r0']=case.pop('nand_init_return')

                assert int(case['main_control_pointer'],16)==0x6009a1f8,'Main control pointer corrupted: stop'

        elif bbt:

            assert int(case['bbt_return'],16) in (0,0xffffffff)
