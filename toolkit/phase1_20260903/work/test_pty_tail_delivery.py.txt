"""Harmless synthetic WinPTY test. Never opens the Rockchip console or USB."""
import os
import sys
import time
import threading
from pathlib import Path
from winpty import PtyProcess


def child():
    for _ in range(4):
        for percent in range(100):
            sys.stdout.write('\x1b[?25l\r\x1b[1ARead Sector to file (%d%%)\x1b[0K\r\n'%percent)
            sys.stdout.flush()
        sys.stdout.write('\r\nRead Sector to file (100%)\r\nREADY>')
        sys.stdout.flush()
        input()


def test(blocking, direct=False):
    os.environ['PYWINPTY_BLOCK']=str(int(blocking))
    from direct_winpty_console import DirectConsole
    factory=DirectConsole if direct else PtyProcess
    proc=factory.spawn([sys.executable,str(Path(__file__).resolve()),'--child'],backend=1,dimensions=(40,160))
    chunks=[]
    stop=threading.Event()
    def read():
        try:
            while proc.isalive():
                value=proc.read(4096)
                if value:
                    chunks.append(value)
        except EOFError:
            pass
        finally:
            stop.set()
    thread=threading.Thread(target=read,daemon=True)
    thread.start()
    passed=0
    try:
        for expected in range(1,5):
            deadline=time.monotonic()+12
            while time.monotonic()<deadline:
                if ''.join(chunks).count('READY>')>=expected:
                    passed+=1
                    proc.write('\r\n')
                    break
                if stop.is_set():
                    break
                time.sleep(.02)
            else:
                break
        print('blocking',blocking,'ready tails delivered',passed,'of 4',flush=True)
    finally:
        if proc.isalive():
            proc.terminate(force=True)
        stop.wait(1)
    return passed


if __name__=='__main__':
    if '--child' in sys.argv:
        child()
    else:
        print('Blocking result:',test(True))
        print('Nonblocking result:',test(False))
        print('Direct nonblocking result:',test(False,direct=True))
